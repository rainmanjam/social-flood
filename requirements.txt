# Caching
# HELD at 7.x deliberately. redis-py 8.x changes DEFAULT_RESP_VERSION from 2
# to 3, so clients negotiate RESP3 by default and fail against Redis servers
# older than 6. app/core/redis_manager.py:72 calls aioredis.from_url() without
# protocol=, and the suite runs with REDIS_URL=None, so nothing here would
# catch it. There are no advisories against 7.1.0, so the bump buys nothing.
# Unblock by passing protocol=2 in redis_manager.py, then move to 8.x.
redis[hiredis]==7.1.0
